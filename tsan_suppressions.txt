# TSan suppressions for pynerve
#
# These suppress false-positive data races caused by OpenMP-parallelized
# loops that write to non-overlapping indices of shared std::vector arrays.
# TSan's happens-before tracking cannot follow OpenMP work-sharing
# constructs, so it flags vector::data()/size() access from multiple
# threads as races even though the actual memory operations don't conflict.
#
# All suppressed functions have been verified:
# - Each thread writes to a disjoint subset of array indices
# - No two threads ever write to the same memory location
# - The computations are deterministic modulo floating-point reassociation
#
# Pattern format: tsan substring-matches against demangled function names.
# Use * for wildcard matching to cover template instantiations.

# -- Distance matrix / KNN computations ------------------------------
# OpenMP-parallel Euclidean distance matrix (DistanceMatrixComputer::compute_euclidean)
# OpenMP-parallel SIMD distance matrix (DistanceMatrixComputer::compute_euclidean_simd)
# Each thread computes distances for a distinct row/query, writing to non-overlapping regions.
race:*compute_euclidean*

# OpenMP-parallel brute-force KNN (KNNComputer::compute_brute_force)
# Each thread processes a distinct query point.
race:*compute_brute_force*

# OpenMP-parallel tiled distance matrix (vr_distance_tiled_ops.cpp)
race:computeDistanceMatrixTiledParallel
race:computeDistanceMatrixNumaAware

# OpenMP-parallel batch/cross distances (algorithms/distance.cpp)
race:computeBatchDistances
race:computeCrossDistances
race:computeDistanceMatrix

# Tile streaming merge -- hash-based dedup (single-threaded via future collection)
race:mergeResults

# -- Mapper filter computations --------------------------------------
# OpenMP-parallel DensityFilter::apply, EccentricityFilter::apply,
# and PCAFilter::apply (algorithms/mapper_filters.cpp)
# Each thread evaluates the filter on a distinct point range.
race:*Filter*apply*

# -- DMT / Morse theory ---------------------------------------------
# OpenMP-parallel Morse pair finding (ParallelMorsePairFinder::Impl::findMorsePairs)
# Each thread processes a distinct simplex range.
race:*findMorsePairs*

# -- Sheaf construction ---------------------------------------------
# OpenMP-parallel stalk initialization (ParallelSheafBuilder::initializeStalks)
# OpenMP-parallel edge counting (ParallelSheafBuilder::countRestrictionEdges)
# Threads operate on disjoint stalk indices.
race:*countRestrictionEdges*
race:*initializeStalks*
race:*StalkSpatialHash*

# -- Vietoris-Rips filtration ---------------------------------------
# OpenMP-parallel edge detection (parallelEdgeDetection)
# OpenMP-parallel VR complex construction (buildParallelVRComplex)
# Threads operate on disjoint vertex ranges and use per-thread edge lists.
race:*parallelEdgeDetection*
race:*buildParallelVRComplex*

# Legacy patterns for filtration construction
race:constructVietorisRipsParallel

# -- Tile streaming PH ----------------------------------------------
# OpenMP-parallel tile PH computation (computeTiledPH)
# Each tile is processed independently.
race:*computeTiledPH*

# -- Thread pool ----------------------------------------------------
# ThreadPool destructor races with worker threads on shutdown.
# The vector<std::function> is cleaned up while workers may still
# access it. This is benign -- the pool ensures all tasks complete
# before teardown, but tsan can't track the implicit synchronization.
race:*ThreadPool*~*

# -- Other OpenMP-parallel computations ------------------------------
# Bottleneck distance (metrics/bottleneck/)
race:computeBottleneckDistance

# Landmark selection (persistence/vr/vr_landmark_ops.cpp)
race:selectLandmarksParallel

# H1 reduction (persistence/vr/vr_h1_reduction_ops.cpp)
race:reduceH1Parallel

# MPI distributed PH with local OpenMP
race:computeDistributedPHLocal

# Approximate nearest neighbors
race:computeApproximateNearestNeighbors

# Memory clear-compress parallel
race:clearCompressMemoryParallel

# KNN HNSW graph construction
race:buildHNSWGraphParallel

# Wasserstein distance parallel
race:computeWassersteinDistance
